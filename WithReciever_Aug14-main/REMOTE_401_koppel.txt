float minTemp = 20.0;
float maxTemp = 30.0;
unsigned long sendDataPrevMillis = 0;

void setup() {
  Serial.begin(115200);
  Wire.begin(OLED_SDA, OLED_SCL);
  irsend.begin();
  
  pinMode(LED_PIN, OUTPUT);
  pinMode(RELAY_PIN, OUTPUT);
  dht.begin();

  if(!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) { 
    Serial.println("OLED Failed");
    for(;;); 
  }

  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  while (WiFi.status() != WL_CONNECTED) { delay(500); Serial.print("."); }
  Serial.println("\nWiFi Connected");

  config.signer.tokens.legacy_token = DATABASE_SECRET;
  config.database_url = DATABASE_URL;
  Firebase.begin(&config, &auth);
  Firebase.reconnectWiFi(true);

  if (!Firebase.RTDB.beginStream(&stream, "/Room_1")) {
    Serial.println("Stream Error: " + stream.errorReason());
  } else {
    Serial.println("Stream OK - Listening to /Room_1");
  }
}

void loop() {
  // --- REAL-TIME STREAM LISTENER ---
  if (Firebase.ready() && Firebase.RTDB.readStream(&stream)) {
    if (stream.streamAvailable()) {
      String path = stream.dataPath(); 
      
      // Update Thresholds via Stream
      if (path == "/minTemp") {
        minTemp = stream.floatData();
        Serial.printf("New Min: %.1f\n", minTemp);
      } 
      else if (path == "/maxTemp") {
        maxTemp = stream.floatData();
        Serial.printf("New Max: %.1f\n", maxTemp);
      }
      // Update AC Commands via Stream
      else if (path.endsWith("ac_command")) {
        handleACCommand(stream.stringData());
      }
      // Handle Multi-update JSON
      else if (path == "/") {
        FirebaseJson &json = stream.jsonObject();
        FirebaseJsonData jsonData;
        if (json.get(jsonData, "ac_command")) handleACCommand(jsonData.stringValue);
        if (json.get(jsonData, "minTemp")) minTemp = jsonData.floatValue;
        if (json.get(jsonData, "maxTemp")) maxTemp = jsonData.floatValue;
      }
    }
  }

  // --- SENSOR & OLED LOGIC ---
  float t = dht.readTemperature();
  bool isAlarm = (t < minTemp || t > maxTemp);
  digitalWrite(LED_PIN, isAlarm ? HIGH : LOW);

  display.clearDisplay();
  display.fillScreen(SSD1306_WHITE); 
  display.setTextColor(SSD1306_BLACK); 
  display.setTextSize(1);
  display.setCursor(5, 5);
  display.printf("MIN:%.1f MAX:%.1f", minTemp, maxTemp); 
  display.setCursor(25, 30);
  display.setTextSize(2);
  if (isnan(t)) display.print("ERR");
  else { display.print(t, 1); display.print(" C"); }
  display.display();

  // --- DATA UPLOAD (Every 10s) ---
  if (millis() - sendDataPrevMillis > 10000) {
    sendDataPrevMillis = millis();
    FirebaseJson updateData;
    updateData.add("temperature", t);
    updateData.add("status", isAlarm ? "ALARM" : "NORMAL");
    Firebase.RTDB.updateNode(&fbdo, "/Room_1", &updateData);
  }
}

void handleACCommand(String cmd) {
  if (cmd == "IDLE") return;

  Serial.println("Command Received: " + cmd);

  if (cmd == "ON") {
    Serial.println(">>> SENDING IR: ON");
    irsend.sendRaw(rawOn, 149, kFrequency);
    Firebase.RTDB.setString(&fbdo, "/Room_1/ac_command", "IDLE");
  } 
  else if (cmd == "OFF") {
    Serial.println(">>> SENDING IR: OFF");
    irsend.sendRaw(rawOff, 149, kFrequency);
    Firebase.RTDB.setString(&fbdo, "/Room_1/ac_command", "IDLE");
  }
}
